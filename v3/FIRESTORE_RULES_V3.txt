// 230MATCH V3 Stage32.1 전용 Firestore 규칙 예시
// Firebase Console > Firestore Database > 규칙에서 기존 규칙과 병합 후 게시하세요.
// 기존 users/{uid} 문서의 role 값이 admin/developer/operator/manager/staff 중 하나여야 씁니다.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }
    function myProfile() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function isApprovedOperator() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && myProfile().approved != false
        && myProfile().role in ['admin', 'developer', 'operator', 'manager', 'staff', 'tournament_admin', 'tournament_operator'];
    }

    match /v3TournamentRooms/{roomId} {
      allow read: if signedIn();
      allow create, update: if isApprovedOperator()
        && request.resource.data.schemaVersion == 3
        && request.resource.data.roomId == roomId
        && request.resource.data.lastWriterUid == request.auth.uid;
      allow delete: if signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && myProfile().role in ['admin', 'developer'];
    }
  }
}
